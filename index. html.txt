<script>
    // =========================================================================
    // 1. CẤU HÌNH KEY ADMIN & DỮ LIỆU BAN ĐẦU
    // =========================================================================
    const ADMIN_KEY = "123456"; // Mã Admin của bạn

    const defaultQuizData = [
        {
            question: "Từ nào sau đây dùng để chỉ 'Dòng điện' trong tiếng Anh kỹ thuật?",
            options: ["Voltage", "Current", "Resistance", "Power"],
            answer: 1
        },
        {
            question: "Thiết bị 'Transformer' trong ngành điện có nghĩa là gì?",
            options: ["Máy biến áp", "Động cơ điện", "Máy phát điện", "Tụ điện"],
            answer: 0
        },
        {
            question: "Linh kiện điện tử 'MOSFET' thuộc loại linh kiện nào?",
            options: ["Transistor BJT", "Transistor trường (FET)", "Diode chỉnh lưu", "Relay"],
            answer: 1
        },
        {
            question: "'Direct Current (DC)' nghĩa là gì?",
            options: ["Dòng điện xoay chiều", "Dòng điện một chiều", "Điện áp cao thế", "Công suất tiêu thụ"],
            answer: 1
        },
        {
            question: "Ký hiệu 'PWM' viết tắt của cụm từ nào?",
            options: ["Pulse Width Modulation", "Power Wave Management", "Phase Width Measurement", "Peak Watt Modulation"],
            answer: 0
        }
    ];

    // Lấy câu hỏi từ bộ nhớ trình duyệt, nếu chưa có thì dùng bộ mặc định
    let quizData = JSON.parse(localStorage.getItem('quizDataApp')) || defaultQuizData;

    // =========================================================================
    // 2. LOGIC LÀM BÀI VÀ XÁO TRỘN CÂU HỎI
    // =========================================================================
    let currentQuestions = [];
    let currentQuestionIndex = 0;
    let userAnswers = [];
    let studentName = "";
    let studentSTT = "";

    function shuffleArray(array) {
        let arr = [...array];
        for (let i = arr.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            [arr[i], arr[j]] = [arr[j], arr[i]];
        }
        return arr;
    }

    function showScreen(screenId) {
        document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));
        document.getElementById(screenId).classList.add('active');
    }

    function startQuiz() {
        if (quizData.length === 0) {
            alert("Hiện chưa có câu hỏi nào trong đề! Vui lòng vào Quản trị để thêm.");
            return;
        }

        studentName = document.getElementById('student-name').value.trim();
        studentSTT = document.getElementById('student-stt').value.trim();
        const errorMsg = document.getElementById('error-message');

        if (!studentName || !studentSTT) {
            errorMsg.style.display = 'block';
            return;
        }
        errorMsg.style.display = 'none';

        // Xáo trộn CÂU HỎI & Xáo trộn ĐÁP ÁN bên trong từng câu
        currentQuestions = shuffleArray(quizData).map(q => {
            const correctAnswerText = q.options[q.answer];
            const shuffledOptions = shuffleArray(q.options);
            const newAnswerIndex = shuffledOptions.indexOf(correctAnswerText);
            return {
                question: q.question,
                options: shuffledOptions,
                answer: newAnswerIndex
            };
        });

        currentQuestionIndex = 0;
        userAnswers = new Array(currentQuestions.length).fill(null);

        document.getElementById('quiz-user-display').innerText = `STT: ${studentSTT} - ${studentName}`;
        showScreen('quiz-screen');
        loadQuestion();
    }

    function loadQuestion() {
        const q = currentQuestions[currentQuestionIndex];
        const quizContent = document.getElementById('quiz-content');
        const progress = document.getElementById('quiz-progress');
        const nextBtn = document.getElementById('next-btn');

        progress.innerText = `Câu ${currentQuestionIndex + 1}/${currentQuestions.length}`;

        let optionsHTML = '';
        q.options.forEach((opt, idx) => {
            const isChecked = userAnswers[currentQuestionIndex] === idx ? 'checked' : '';
            optionsHTML += `
                <li class="option-item">
                    <label class="option-label">
                        <input type="radio" name="option" value="${idx}" ${isChecked} onchange="selectOption(${idx})">
                        <span>${opt}</span>
                    </label>
                </li>
            `;
        });

        quizContent.innerHTML = `
            <div class="question-box">
                <div class="question-text">Câu ${currentQuestionIndex + 1}: ${q.question}</div>
                <ul class="options-list">${optionsHTML}</ul>
            </div>
        `;

        nextBtn.innerText = (currentQuestionIndex === currentQuestions.length - 1) ? "Nộp Bài" : "Câu Tiếp Theo";
    }

    function selectOption(index) {
        userAnswers[currentQuestionIndex] = index;
    }

    function nextQuestion() {
        if (userAnswers[currentQuestionIndex] === null) {
            alert("Vui lòng chọn 1 đáp án trước khi chuyển câu!");
            return;
        }

        if (currentQuestionIndex < currentQuestions.length - 1) {
            currentQuestionIndex++;
            loadQuestion();
        } else {
            finishQuiz();
        }
    }

    function finishQuiz() {
        let score = 0;
        currentQuestions.forEach((q, idx) => {
            if (userAnswers[idx] === q.answer) score++;
        });

        document.getElementById('res-name').innerText = studentName;
        document.getElementById('res-stt').innerText = studentSTT;
        document.getElementById('res-score').innerText = `${score} / ${currentQuestions.length}`;

        showScreen('result-screen');
    }

    function restartQuiz() {
        showScreen('start-screen');
    }

    // =========================================================================
    // 3. LOGIC XÁC THỰC KEY VÀ CHỈNH SỬA ĐỀ THI (ADMIN)
    // =========================================================================
    function promptAdminKey() {
        const inputKey = prompt("Nhập mã Key Admin để chỉnh sửa câu hỏi:");
        if (inputKey === ADMIN_KEY) {
            renderAdminQuestions();
            showScreen('admin-screen');
        } else if (inputKey !== null) {
            alert("Sai Key Admin!");
        }
    }

    function saveQuestionsToStorage() {
        localStorage.setItem('quizDataApp', JSON.stringify(quizData));
    }

    function renderAdminQuestions() {
        const listContainer = document.getElementById('admin-questions-list');
        document.getElementById('total-q-count').innerText = quizData.length;
        listContainer.innerHTML = '';

        quizData.forEach((q, idx) => {
            listContainer.innerHTML += `
                <div class="question-item-admin">
                    <span><strong>Câu ${idx + 1}:</strong> ${q.question}</span>
                    <button class="btn btn-danger" onclick="deleteQuestion(${idx})">Xóa</button>
                </div>
            `;
        });
    }

    function addQuestion() {
        const qText = document.getElementById('new-question').value.trim();
        const opt0 = document.getElementById('opt-0').value.trim();
        const opt1 = document.getElementById('opt-1').value.trim();
        const opt2 = document.getElementById('opt-2').value.trim();
        const opt3 = document.getElementById('opt-3').value.trim();
        const correctAns = parseInt(document.getElementById('correct-opt').value);

        if (!qText || !opt0 || !opt1 || !opt2 || !opt3) {
            alert("Vui lòng điền đầy đủ câu hỏi và cả 4 đáp án!");
            return;
        }

        quizData.push({
            question: qText,
            options: [opt0, opt1, opt2, opt3],
            answer: correctAns
        });

        saveQuestionsToStorage();
        renderAdminQuestions();

        // Clear form input
        document.getElementById('new-question').value = '';
        document.getElementById('opt-0').value = '';
        document.getElementById('opt-1').value = '';
        document.getElementById('opt-2').value = '';
        document.getElementById('opt-3').value = '';
    }

    function deleteQuestion(index) {
        if (confirm("Bạn có chắc chắn muốn xóa câu hỏi này?")) {
            quizData.splice(index, 1);
            saveQuestionsToStorage();
            renderAdminQuestions();
        }
    }
</script>
